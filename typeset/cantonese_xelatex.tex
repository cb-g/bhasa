\subsection{Vowels}

\begin{xltabular}{\textwidth}{lL}
  \ipa{i:} & long close front unrounded vowel \\
  \ipa{y:} & long close front rounded vowel \\
  \ipa{u:} & long close back rounded vowel \\
  \ipa{e} & close-mid front unrounded vowel (short, in -eng, -ek only) \\
  \ipa{E:} & long open-mid front unrounded vowel \\
  \ipa{\oe:} & long open-mid front rounded vowel \\
  \ipa{@} & mid central vowel (short, in -eung, -euk only) \\
  \ipa{8} & close-mid central rounded vowel (short, in -eoi, -eon, -eot only) \\
  \ipa{o} & close-mid back rounded vowel (short, in -ong, -ok only) \\
  \ipa{O:} & long open-mid back rounded vowel \\
  \ipa{5} & short near-open central vowel \\
  \ipa{a:} & long open front unrounded vowel \\
\end{xltabular}

\subsection{Consonants}

\begin{xltabular}{\textwidth}{lL}
  \ipa{m} & voiced bilabial nasal \\
  \ipa{n} & voiced alveolar nasal \\
  \ipa{N} & voiced velar nasal \\
  \ipa{p} & voiceless bilabial plosive (unaspirated) \\
  \ipa{p\super h} & voiceless bilabial plosive (aspirated) \\
  \ipa{t} & voiceless alveolar plosive (unaspirated) \\
  \ipa{t\super h} & voiceless alveolar plosive (aspirated) \\
  \ipa{k} & voiceless velar plosive (unaspirated) \\
  \ipa{k\super h} & voiceless velar plosive (aspirated) \\
  \ipa{kw} & voiceless labialized velar plosive (unaspirated) \\
  \ipa{k\super hw} & voiceless labialized velar plosive (aspirated) \\
  \ipa{ts} & voiceless alveolar affricate (unaspirated) \\
  \ipa{ts\super h} & voiceless alveolar affricate (aspirated) \\
  \ipa{f} & voiceless labiodental fricative \\
  \ipa{s} & voiceless alveolar fricative \\
  \ipa{h} & voiceless glottal fricative \\
  \ipa{l} & voiced alveolar lateral approximant \\
  \ipa{w} & voiced labio-velar approximant \\
  \ipa{j} & voiced palatal approximant \\
\end{xltabular}
