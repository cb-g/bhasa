\subsection{Vowels}

\begin{xltabular}{\textwidth}{lL}
  \ipa{i} & close front unrounded vowel \\
  \ipa{e} & close-mid front unrounded vowel (merged with \ipa{E}) \\
  \ipa{a} & open front unrounded vowel \\
  \ipa{o} & close-mid back rounded vowel \\
  \ipa{u} & close back rounded vowel \\
  \ipa{2} & close-mid back unrounded vowel \\
  \ipa{1} & close back unrounded vowel \\
\end{xltabular}

\subsection{Consonants}

\begin{xltabular}{\textwidth}{lL}
  \ipa{m} & voiced bilabial nasal \\
  \ipa{n} & voiced alveolar nasal \\
  \ipa{N} & voiced velar nasal \\
  \ipa{p} & voiceless bilabial plosive (lenis) \\
  \ipa{p\super h} & voiceless bilabial plosive (aspirated) \\
  \ipa{p:} & voiceless bilabial plosive (fortis/tense) \\
  \ipa{t} & voiceless alveolar plosive (lenis) \\
  \ipa{t\super h} & voiceless alveolar plosive (aspirated) \\
  \ipa{t:} & voiceless alveolar plosive (fortis/tense) \\
  \ipa{k} & voiceless velar plosive (lenis) \\
  \ipa{k\super h} & voiceless velar plosive (aspirated) \\
  \ipa{k:} & voiceless velar plosive (fortis/tense) \\
  \ipa{t\textctc} & voiceless alveolo-palatal affricate (lenis) \\
  \ipa{t\textctc\super h} & voiceless alveolo-palatal affricate (aspirated) \\
  \ipa{t\textctc:} & voiceless alveolo-palatal affricate (fortis/tense) \\
  \ipa{s} & voiceless alveolar fricative (lenis) \\
  \ipa{s:} & voiceless alveolar fricative (fortis/tense) \\
  \ipa{h} & voiceless glottal fricative \\
  \ipa{l\*r} & voiced alveolar lateral approximant \textasciitilde{} flap \\
  \ipa{w} & voiced labio-velar approximant \\
  \ipa{j} & voiced palatal approximant \\
  \ipa{\textturnm} & voiced velar approximant \\
\end{xltabular}
