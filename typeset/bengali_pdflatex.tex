\begin{xltabular}{\textwidth}{lL}
  \ipa{O} & open-mid back rounded vowel \\
  \ipa{o} & close-mid back rounded vowel \\
  \ipa{5} & near-open central vowel \\
  \ipa{\"a} & open central unrounded vowel \\
  \ipa{i} & close front unrounded vowel \\
  \ipa{u} & close back rounded vowel \\
  \ipa{e} & close-mid front unrounded vowel \\
  \ipa{\ae} & near-open front unrounded vowel \\
  \ipa{\~a} & nasalized vowel (e.g. nasalized \ipa{\"a}) \\
  \ipa{oi} & diphthong: close-mid back rounded plus close front unrounded vowel \\
  \ipa{ou} & diphthong: close-mid back rounded plus close back rounded vowel \\
\end{xltabular}
