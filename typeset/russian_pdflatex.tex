\subsection{Vowels}

\begin{xltabular}{\textwidth}{lL}
  \ipa{i} & close front unrounded vowel \\
  \ipa{1} & close central unrounded vowel (disputed as separate from /i/) \\
  \ipa{u} & close back rounded vowel \\
  \ipa{e} & close-mid front unrounded vowel \\
  \ipa{o} & close-mid back rounded vowel \\
  \ipa{a} & open central unrounded vowel \\
\end{xltabular}

  & \\
  \multicolumn{2}{l}{\textbf{Note:} Vowel reduction occurs in unstressed syllables. Unstressed /o/ and /a/} \\
  \multicolumn{2}{l}{merge to [\ipa{5}] or [\ipa{@}]; unstressed /e/ merges with /i/. The symbol [\ipa{1}] appears} \\
