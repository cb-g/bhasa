\subsection{Vowels}

\begin{xltabular}{\textwidth}{lL}
  \ipa{i} & close front unrounded vowel \\
  \ipa{y} & close front rounded vowel \\
  \ipa{1} & close back unrounded vowel \\
  \ipa{u} & close back rounded vowel \\
  \ipa{e} & close-mid front unrounded vowel \\
  \ipa{\o} & close-mid front rounded vowel \\
  \ipa{a} & open front unrounded vowel \\
  \ipa{o} & close-mid back rounded vowel \\
\end{xltabular}

\subsection{Consonants}

\begin{xltabular}{\textwidth}{lL}
  \ipa{m} & voiced bilabial nasal \\
  \ipa{n} & voiced alveolar nasal \\
  \ipa{p} & voiceless bilabial plosive \\
  \ipa{b} & voiced bilabial plosive \\
  \ipa{t} & voiceless alveolar plosive \\
  \ipa{d} & voiced alveolar plosive \\
  \ipa{k} & voiceless velar plosive (palatal [c] before front vowels) \\
  \ipa{g} & voiced velar plosive (palatal [\textbardotlessj] before front vowels) \\
  \ipa{\t{tS}} & voiceless postalveolar affricate \\
  \ipa{\t{dZ}} & voiced postalveolar affricate \\
  \ipa{f} & voiceless labiodental fricative \\
  \ipa{v} & voiced labiodental fricative \\
  \ipa{s} & voiceless alveolar fricative \\
  \ipa{z} & voiced alveolar fricative \\
  \ipa{S} & voiceless postalveolar fricative \\
  \ipa{Z} & voiced postalveolar fricative (in loanwords) \\
  \ipa{h} & voiceless glottal fricative \\
  \ipa{l} & voiced alveolar lateral approximant (velarized [\textltilde l] before back vowels) \\
  \ipa{\*r} & voiced alveolar tap/trill \\
  \ipa{j} & voiced palatal approximant \\
\end{xltabular}
